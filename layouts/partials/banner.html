{{- /* Renders a photo banner. name: image filename, looked up in content/banners/ first, then content/field-notes/ images: (style "multi" only) list of {image, rotate?} shown side by side, left to right style: "hero" = full-bleed cropped band at top of page "strip" = full-bleed cropped band between sections "fit" = full-bleed band; whole photo height shown, blurred edges fill the sides "multi" = several photos side by side in one band id: optional anchor id so nav links land on the banner */ -}} {{- $style := .style | default "strip" -}} {{- $id := .id -}} {{- if eq $style "multi-fit" -}}
{{- range .images -}} {{- $imgName := .image -}} {{- $rot := "" -}} {{- with .rotate -}}{{- $rot = printf " r%d" (sub 360 (int .)) -}}{{- end -}} {{- $r := "" -}} {{- with site.GetPage "/banners" -}}{{- $r = .Resources.GetMatch $imgName -}}{{- end -}} {{- if not $r -}} {{- with site.GetPage "/field-notes" -}}{{- $r = .Resources.GetMatch $imgName -}}{{- end -}} {{- end -}} {{- with $r -}} {{- with .Resize (printf "x720 q80 webp%s" $rot) }} {{- end -}} {{- end -}} {{- end }}
{{- else if eq $style "multi" -}} {{- $mobile := .mobile -}}
{{- range .images -}} {{- $imgName := .image -}} {{- $fit := .fit -}} {{- $noblur := eq .blur false -}} {{- $anchor := .anchor -}} {{- $rot := "" -}} {{- with .rotate -}}{{- $rot = printf " r%d" (sub 360 (int .)) -}}{{- end -}} {{- $r := "" -}} {{- with site.GetPage "/banners" -}}{{- $r = .Resources.GetMatch $imgName -}}{{- end -}} {{- if not $r -}} {{- with site.GetPage "/field-notes" -}}{{- $r = .Resources.GetMatch $imgName -}}{{- end -}} {{- end -}} {{- with $r -}} {{- if and $fit $noblur -}} {{- with .Resize (printf "x720 q80 webp%s" $rot) }} {{- end -}} {{- else if $fit -}} {{- $bg := .Fill (printf "842x720 q40 webp%s" $rot) -}} {{- with .Resize (printf "x720 q80 webp%s" $rot) }} {{- end -}} {{- else -}} {{- with .Fill (printf "842x720 q70 webp%s%s" $rot (cond (ne $anchor nil) (printf " %s" $anchor) "")) }} {{- end -}} {{- end -}} {{- end -}} {{- end }} {{- with $mobile -}} {{- $mr := "" -}} {{- with site.GetPage "/banners" -}}{{- $mr = .Resources.GetMatch $mobile -}}{{- end -}} {{- if not $mr -}}{{- with site.GetPage "/field-notes" -}}{{- $mr = .Resources.GetMatch $mobile -}}{{- end -}}{{- end -}} {{- with $mr -}}{{- with .Fill "1280x360 q70 webp" }} {{- end -}}{{- end -}} {{- end }}
{{- else -}} {{- $name := .name -}} {{- $res := "" -}} {{- with site.GetPage "/banners" -}}{{- $res = .Resources.GetMatch $name -}}{{- end -}} {{- if not $res -}} {{- with site.GetPage "/field-notes" -}}{{- $res = .Resources.GetMatch $name -}}{{- end -}} {{- end -}} {{- with $res -}} {{- if eq $style "hero" -}} {{- with .Fill "1280x360 q75 webp" }}
{{- end -}} {{- else if eq $style "fit" -}} {{- $bg := .Fill "1280x360 q40 webp" -}} {{- with .Resize "x720 q80 webp" }}
{{- end -}} {{- else -}} {{- with .Fill "1280x360 q70 webp" }}
{{- end -}} {{- end -}} {{- end -}} {{- end -}}
